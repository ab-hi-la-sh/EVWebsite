{{- /* FAQPage schema, emitted only when a page opts in with faqSchema: true. Two sources, in priority order, so the schema always matches what the page actually shows: 1. `.Params.faqs` -- blog articles define FAQs in front matter and the layout renders the accordion from it (after .Content), so the questions are NOT in .Content and must be read from the param directly. 2. Rendered
in .Content -- product/FAQ pages that author the accordion inline. Reading the rendered content keeps schema and visible content in lockstep (the old hand-maintained JSON had already drifted: nopii/faqs listed 14 questions but only 12 in schema; vault/faqs had 10 visible, 9 in schema). Google requires schema to match visible content. Opt-in matters: not every
is an FAQ (nopii/overview's "Under the Hood" list is feature descriptions, which must not be published as Questions). */ -}} {{- if .Params.faqSchema -}} {{- $qa := slice -}} {{- with .Params.faqs -}} {{- range . -}} {{- $q := .q | htmlUnescape | strings.TrimSpace -}} {{- $a := .a | htmlUnescape | strings.TrimSpace -}} {{- if and $q $a -}} {{- $qa = $qa | append (dict "@type" "Question" "name" $q "acceptedAnswer" (dict "@type" "Answer" "text" $a)) -}} {{- end -}} {{- end -}} {{- else -}} {{- range findRE `(?s)` .Content -}} {{- $block := . -}} {{- $q := "" -}}{{- $a := "" -}} {{- range first 1 (findRE `(?s)]*>.*?` $block) -}} {{- $q = . | replaceRE `<[^>]*>` "" | htmlUnescape | strings.TrimSpace -}} {{- end -}} {{- range first 1 (findRE `(?s)
.*?
` $block) -}} {{- $a = . | replaceRE `<[^>]*>` "" | htmlUnescape | strings.TrimSpace -}} {{- end -}} {{- if and $q $a -}} {{- $qa = $qa | append (dict "@type" "Question" "name" $q "acceptedAnswer" (dict "@type" "Answer" "text" $a)) -}} {{- end -}} {{- end -}} {{- end -}} {{- with $qa -}} {{- end -}} {{- end -}}
